private async void OnObtenerUbicacionClicked(object sender, EventArgs e)
{
try
{
// Pedir permiso
var status = await
Permissions.CheckStatusAsync<Permissions.LocationWhenInUse>();
if (status != PermissionStatus.Granted)
status = await
Permissions.RequestAsync<Permissions.LocationWhenInUse>();
if (status != PermissionStatus.Granted)
{
lblEstado.Text = "Permiso denegado";
return;
}
// Obtener ubicación
lblEstado.Text = "Buscando ubicación...";
var location = await Geolocation.GetLocationAsync(
new GeolocationRequest(GeolocationAccuracy.Medium));
if (location != null)
{
lblLatitud.Text = $"Latitud: {location.Latitude:F6}";
lblLongitud.Text = $"Longitud: {location.Longitude:F6}";
lblEstado.Text = "✅ Ubicación obtenida";
}
}
catch (Exception ex)
{
lblEstado.Text = $"Error: {ex.Message}";
}
}

Resultado esperado: Al presionar el botón, aparecen las
coordenadas del celular/emulador.

private async void OnSacarFotoClicked(object sender, EventArgs e)
{
try
{
if (!MediaPicker.IsCaptureSupported)
{
await DisplayAlert("Error", "Este dispositivo no tiene cámara",
"OK");
return;
}
var photo = await MediaPicker.CapturePhotoAsync();
if (photo != null)
imgFoto.Source = ImageSource.FromFile(photo.FullPath);
}
catch (Exception ex)
{
await DisplayAlert("Error", ex.Message, "OK");
}
}
private async void OnElegirGaleriaClicked(object sender, EventArgs e)
{
try
{
var photo = await MediaPicker.PickPhotoAsync();
if (photo != null)
imgFoto.Source = ImageSource.FromFile(photo.FullPath);

Ejercicio 3: Compartir un texto
Crear una pantalla con un Entry para escribir un mensaje y un botón
para compartirlo.
XAML:
<StackLayout Padding="20" Spacing="15" VerticalOptions="Center">
<Label Text="📤 Compartir" FontSize="24" FontAttributes="Bold"
HorizontalOptions="Center" />
<Entry x:Name="txtMensaje" Placeholder="Escribí un mensaje para
compartir..." />
<Button Text="Compartir" Clicked="OnCompartirClicked"
BackgroundColor="#512BD4" TextColor="White" />
</StackLayout>

Code-behind:
private async void OnCompartirClicked(object sender, EventArgs e)
{
if (string.IsNullOrWhiteSpace(txtMensaje.Text))
{
await DisplayAlert("Error", "Escribí algo para compartir", "OK");
return;
}
await Share.RequestAsync(new ShareTextRequest
{
Text = txtMensaje.Text,
Title = "Compartir mensaje"
});
}

Resultado esperado: Al presionar compartir, se abre el menú del
sistema para enviar por WhatsApp, email, etc.

Ejercicio 4: Detectar conectividad
Crear una pantalla que muestre si hay internet o no, y que se actualice
automáticamente.
XAML:
<StackLayout Padding="20" Spacing="15" VerticalOptions="Center">
<Label Text="📶 Conectividad" FontSize="24" FontAttributes="Bold"
HorizontalOptions="Center" />
<Frame x:Name="frameEstado" Padding="20" CornerRadius="10"
HorizontalOptions="Center">
<Label x:Name="lblConectividad" FontSize="18"
HorizontalOptions="Center" />
</Frame>
<Button Text="Verificar Ahora" Clicked="OnVerificarClicked"
BackgroundColor="#512BD4" TextColor="White" />
</StackLayout>

Code-behind:
public ConectividadPage()
{
InitializeComponent();
VerificarConectividad();
// Escuchar cambios automáticamente
Connectivity.ConnectivityChanged += (s, e) => VerificarConectividad();
}
private void VerificarConectividad()
{
if (Connectivity.Current.NetworkAccess == NetworkAccess.Internet)
{
lblConectividad.Text = "✅ Conectado a internet";
frameEstado.BackgroundColor = Color.FromArgb("#E8F5E9");
}
else
{
lblConectividad.Text = "❌ Sin conexión";
frameEstado.BackgroundColor = Color.FromArgb("#FFEBEE");
}
}
private void OnVerificarClicked(object sender, EventArgs e)
{
VerificarConectividad();
}

💡 Tips
• Cada ejercicio puede ser una página separada con navegación, o
todo en una sola app con tabs
• En el emulador: la cámara simula una imagen, el GPS usa una
ubicación ficticia, la vibración no se siente pero no da error

• No te preocupes por el diseño perfecto — lo importante es que
funcione
